# Age distribution by mortality (stacked)
plt.figure(figsize=(8,5))
sns.histplot(data=df, x="age", hue="y", bins=30, multiple="stack", kde=False)
plt.title("Age Distribution by Hospital Mortality (y)")
plt.xlabel("Age")
plt.ylabel("Count")
plt.tight_layout()
plt.show()

# Gender counts
plt.figure(figsize=(7,5))
order = df["gender"].value_counts().index
sns.countplot(data=df, x="gender", order=order)
plt.title("Gender Counts")
#Visualisation

plt.xlabel("Gender")
plt.ylabel("Count")
plt.tight_layout()
plt.show()

# Mortality rate by gender
gender_rate = df.groupby("gender")["y"].mean().reset_index().sort_values("y", ascending=False)

plt.figure(figsize=(7,5))
sns.barplot(data=gender_rate, x="gender", y="y", order=gender_rate["gender"])
plt.title("Hospital Mortality Rate by Gender")
plt.xlabel("Gender")
plt.ylabel("Mortality rate (y=1)")
plt.ylim(0, 1)
plt.tight_layout()
plt.show()

# Ethnicity mortality rates
eth_rate = (
    df.groupby("ethnicity")["y"].mean()
      .reset_index()
      .sort_values("y", ascending=False)
)

plt.figure(figsize=(10,5))
sns.barplot(data=eth_rate, x="ethnicity", y="y")
plt.title("Hospital Mortality Rate by Ethnicity")
plt.xlabel("Ethnicity")
plt.ylabel("Mortality rate (y=1)")
plt.ylim(0, 1)
plt.xticks(rotation=45, ha="right")
plt.tight_layout()
plt.show()
